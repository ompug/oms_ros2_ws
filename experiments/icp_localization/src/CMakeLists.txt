cmake_minimum_required(VERSION 3.16)
project(icp_localize LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(libpointmatcher REQUIRED)
find_package(Eigen3 REQUIRED)
add_executable(icp_localize icp_localize.cpp)
target_include_directories(icp_localize PRIVATE ${libpointmatcher_INCLUDE_DIRS})
target_link_libraries(icp_localize PRIVATE ${libpointmatcher_LIBRARIES} Eigen3::Eigen)
